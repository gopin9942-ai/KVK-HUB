allow update: if request.auth != null
  && (resource.data.creatorUid == request.auth.uid
      || resource.data.createdBy == request.auth.uid)
  && request.resource.data.diff(resource.data).affectedKeys()
      .hasOnly(['roomDisplayName', 'roomPhotoURL', 'updatedAt'])
  && request.resource.data.roomDisplayName is string
  && request.resource.data.roomDisplayName.size() > 0
  && request.resource.data.roomDisplayName.size() <= 50
  && request.resource.data.roomPhotoURL is string;
